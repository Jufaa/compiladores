int saldo;

void depositar(int monto) {
    saldo = saldo + monto;
    return;
}

boolean retirar(int monto) {
    if (monto <= 0 || monto > saldo) {
        return false;
    }
    saldo = saldo - monto;
    return true;
}

int main() {
    int i;
    boolean aceptado;
    saldo = 10;
    i = 1;
    while (i <= 4) {
        depositar(i * 3);
        if (i % 2 == 0) {
            aceptado = retirar(i);
        }
        i = i + 1;
    }
    aceptado = retirar(100);
    if (!aceptado) {
        return saldo;
    } else {
        return 99;
    }
}

// Esperado: 34
// Devuelve: 34
